% ===================================================== O PROBLEMA
\begin{frame}{O problema}
  \vspace{0.2em}
  {\small A quebra de louças sanitárias no canteiro gera fragmentos cortantes que
   causam cortes e lacerações em trabalhadores da construção civil.\par}
  \vspace{0.9em}

  \centering
  \begin{tikzpicture}[
      etapa/.style={fill=cinzaclaro,rounded corners=3pt,inner sep=7pt,
                    text width=2.5cm,align=center,font=\scriptsize},
      seta/.style={-{Latex[length=2.4mm]},unbverde,line width=1pt}]
    \node[etapa] (a) {Manuseio, transporte\\ e instalação da peça};
    \node[etapa,right=0.55cm of a] (b) {Quebra do\\ componente cerâmico};
    \node[etapa,right=0.55cm of b] (c) {Fragmento cortante\\ em área de circulação};
    \node[etapa,right=0.55cm of c,fill=unbverdeclaro] (d) {\textbf{Corte ou laceração\\ no trabalhador}};
    \draw[seta] (a) -- (b); \draw[seta] (b) -- (c); \draw[seta] (c) -- (d);
  \end{tikzpicture}

  \vspace{1.1em}
  \raggedright
  \cartao[unbazulclaro]{0.93\textwidth}{%
    {\color{unbazul}\bfseries\small Dupla natureza do problema}\\[0.5em]
    {\footnotesize
    \textbf{Saúde e segurança ocupacional} — risco físico sem controle formal:
    o manuseio de material cerâmico frágil não tem procedimento, EPI específico
    nem sinalização própria.\\[0.35em]
    \textbf{Perdas e desperdício} — custo de material, tempo e reposição da peça;
    é o argumento que sensibiliza o empregador.}
  }
\end{frame}

% ===================================================== ONDE ACONTECE (NOVO)
\begin{frame}{Onde acontece: da chegada da peça à instalação}{Canteiro em fase de acabamento — a quebra pode ocorrer em qualquer das quatro etapas}
  \vspace{0.5em}
  \centering\footnotesize
  \renewcommand{\arraystretch}{1.35}
  \newcolumntype{Y}{>{\RaggedRight\arraybackslash}X}
  \begin{tabularx}{\textwidth}{@{}>{\RaggedRight\arraybackslash}p{3.0cm}YY@{}}
    \toprule
    \textbf{\color{unbazul}Etapa} & \textbf{\color{unbazul}O que acontece} &
    \textbf{\color{unbazul}Como a quebra pode ocorrer}\\
    \midrule
    \textbf{1. Recebimento} & Descarga das peças embaladas do caminhão. &
    Impacto na descarga; empilhamento acima do recomendado.\\
    \textbf{2. Armazenagem} & As peças aguardam a fase de acabamento no estoque. &
    Empilhamento sem berço; estoque em área de circulação.\\
    \textbf{3. Transporte interno} & Deslocamento até o pavimento e o ponto de instalação. &
    Peça sem ponto de pega; piso irregular; entulho no trajeto.\\
    \textbf{4. Instalação} & Posicionamento e fixação no ponto hidráulico. &
    Manobra em espaço confinado; apoio sobre superfície dura.\\
    \bottomrule
  \end{tabularx}

  \vspace{0.9em}
  \raggedright
  \cartao[unbverdeclaro]{0.93\textwidth}{%
    {\footnotesize Nas quatro etapas o desfecho é o mesmo: \textbf{fragmento
     cortante em área de circulação de pessoas}. Identificar qual etapa concentra
     as quebras é um dos objetivos da visita de campo.}
  }
\end{frame}

% ===================================================== DADOS: ACIDENTES
\begin{frame}[shrink=10]{Por que importa: acidentes na construção civil}
  \vspace{0.4em}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.30\textwidth}
      \numerao{\textwidth}{18,9\%}{dos acidentes do setor são \textbf{cortes,
      lacerações e feridas} — atrás apenas das fraturas (25,4\%). Dedos e mãos
      estão entre as partes mais atingidas.}
    \end{column}
    \begin{column}{0.30\textwidth}
      \numerao{\textwidth}{6ª}{posição da construção civil no ranking nacional de
      incidência de acidentes típicos, mesmo após a queda recorde de 2023.}
    \end{column}
    \begin{column}{0.30\textwidth}
      \numerao{\textwidth}{51 s}{é o intervalo médio estimado entre dois acidentes
      de trabalho no Brasil.}
    \end{column}
  \end{columns}

  \vspace{0.9em}
  \cartao[unbazulclaro]{0.93\textwidth}{%
    {\color{unbazul}\bfseries\small E a queda recente de acidentes?}\\[0.4em]
    {\footnotesize
    Os ganhos de 2023 concentram-se em riscos que já têm \textbf{norma, EPI e
    procedimento próprios} — altura, choque elétrico, máquinas. O manuseio de
    cerâmico frágil não tem nenhum dos três, e segue diluído em ``cortes e
    lacerações''. \textbf{A melhora do agregado não enfraquece o problema: torna
    visível o risco que a padronização não alcançou.}}
  }
  \fonte{AEAT/INSS, ano-base 2022 (\textit{apud} Migalhas, 2024); CBIC/SESI, 2025; MPT/Smartlab.}
\end{frame}

% ===================================================== DADOS: PERDAS
\begin{frame}{O custo do outro lado: perdas de material em obra}
  \vspace{0.6em}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.46\textwidth}
      \numerao{\textwidth}{10--30\%}{de todo material adquirido para uma obra é
      descartado como resíduo, mesmo em canteiros com boas práticas.}
      \vspace{1.1em}

      \numerao{\textwidth}{73,2\%}{das empresas do setor apontam o desperdício de
      materiais como o principal problema de gestão em obra.}
    \end{column}
    \begin{column}{0.48\textwidth}
      \cartao{0.9\textwidth}{%
        {\color{unbazul}\bfseries\small Por que este eixo entra}\\[0.6em]
        {\footnotesize
        Componentes frágeis e de acabamento — como as louças sanitárias — são
        ponto crítico de gestão de perdas.\\[0.6em]
        O custo da peça reposta é o \textbf{argumento que sensibiliza o
        empregador} a abrir o canteiro e destinar recurso.\\[0.6em]
        {\color{unbverde}\bfseries O eixo central do projeto permanece a segurança
        do trabalhador.}}
      }
    \end{column}
  \end{columns}
  \fonte{Revista Digital Acadêmico Mundo, 2026; CBIC (\textit{apud} Sienge, 2025).}
\end{frame}

% ===================================================== USUÁRIOS (p.31)
\begin{frame}{Usuários e clientes}{Perguntas da fase de Descoberta — \textit{Design Innovation Methodology Handbook}, p.\,31}
  \vspace{0.4em}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      {\color{unbazul}\bfseries\small Compreender}\\[0.6em]
      {\footnotesize
      \textbf{Quem é o usuário?} Hipótese: não é uma função só — o \textbf{transporte}
      tende a caber ao servente e a \textbf{instalação} ao instalador hidráulico.
      Confirmar em campo define para quem projetamos.\\[0.5em]
      \textbf{Do que precisa?} Concluir a tarefa sem se ferir, sem danificar a peça
      e sem perder tempo — três necessidades que competem entre si.\\[0.5em]
      \textbf{Qual é o contexto?} As quatro etapas do slide anterior.}
    \end{column}
    \begin{column}{0.47\textwidth}
      {\color{unbverde}\bfseries\small Empatizar}\\[0.6em]
      {\footnotesize
      \textbf{Como se comporta?} Carrega sozinho quando o peso permite; a luva é
      usada de forma intermitente, porque reduz a firmeza da pega justamente na
      peça lisa e cara.\\[0.5em]
      \textbf{Como se sente?} A quebra é vivida como falha individual — o receio de
      ser responsabilizado pelo custo inibe o relato.\\[0.5em]
      \textbf{Condições extremas?} Fim de expediente, piso molhado, escada quando o
      elevador está ocupado, pavimento sem luz definitiva.}
    \end{column}
  \end{columns}
  \vspace{0.7em}
  {\scriptsize\color{cinza} As respostas assinaladas como hipótese serão confirmadas
   ou corrigidas na visita de campo.}
\end{frame}

% ===================================================== EXPECTATIVAS
\begin{frame}{O que cada stakeholder espera}
  \vspace{0.6em}\footnotesize
  \renewcommand{\arraystretch}{1.5}
  \begin{tabularx}{\textwidth}{@{}p{4.3cm}X@{}}
    \toprule
    \textbf{\color{unbazul}Trabalhador da construção} &
    Concluir a tarefa sem se ferir e sem ser responsabilizado pela peça quebrada;
    solução simples, que não aumente o tempo do serviço.\\
    \textbf{\color{unbazul}Mestre de obras / gestor} &
    Cumprir o cronograma sem incidente nem retrabalho; qualquer medida precisa
    caber na rotina do canteiro.\\
    \textbf{\color{unbazul}Técnico de segurança (SESMT)} &
    Reduzir indicadores de acidente e sustentar a conformidade com a NR-18;
    espera dado técnico de frequência e tipo de lesão.\\
    \textbf{\color{unbazul}Empregador / construtora} &
    Reduzir custo de reposição, afastamento e passivo; espera ver retorno em
    número antes de destinar recurso.\\
    \textbf{\color{unbazul}Fornecedor de louças} &
    Saber em que ponto da cadeia a peça se rompe; menos avaria reportada é ganho
    comercial dele.\\
    \bottomrule
  \end{tabularx}
\end{frame}

% ===================================================== PROPOSTA DE VALOR

  \cartao[unbverdeclaro]{0.93\textwidth}{%
    {\color{unbverde}\bfseries\small Proposta de valor}\\[0.5em]
    {\footnotesize Reduzir a ocorrência de \textbf{lesão por fragmento cerâmico}
